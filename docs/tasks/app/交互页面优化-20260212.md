class LetterPosition {
  static double getBaselineOffset(String letter) {
    // 大���字母：坐在中线上
    if (letter == letter.toUpperCase()) {
      return 0.5;
    }

    // 小写字母（a,c,e,m,n,o,r,s,u,v,w,x,z）：中线到上线
    if ('acemnorsuvwxz'.contains(letter.toLowerCase())) {
      return 0.375;
    }

    // 高字母（b,d,f,h,k,l,t）：触及顶线
    if ('bdfhklt'.contains(letter.toLowerCase())) {
      return 0.25;
    }

    // 下伸字母（g,j,p,q,y）：延伸到下线
    if ('gjpqy'.contains(letter.toLowerCase())) {
      return 0.625;
    }

    return 0.5;
  }
}

// app_settings_service.dart
import 'package:get_storage/get_storage.dart';

enum PlaybackSpeed { slow, normal, fast }

class AppSettingsService extends GetxService {
  final _storage = GetStorage();
  static const _speedKey = 'playback_speed';

  final playbackSpeed = PlaybackSpeed.normal.obs;

  @override
  void onInit() {
    super.onInit();
    _loadSettings();
  }

  void _loadSettings() {
    final savedSpeed = _storage.read(_speedKey);
    if (savedSpeed != null) {
      playbackSpeed.value = PlaybackSpeed.values.firstWhere(
        (e) => e.toString() == savedSpeed,
        orElse: () => PlaybackSpeed.normal,
      );
    }
  }

  Future<void> setPlaybackSpeed(PlaybackSpeed speed) async {
    playbackSpeed.value = speed;
    await _storage.write(_speedKey, speed.toString());
  }

  ({double chinese, double english}) getSpeedRates() {
    return switch (playbackSpeed.value) {
      PlaybackSpeed.slow => (chinese: 0.5, english: 0.6),
      PlaybackSpeed.normal => (chinese: 0.7, english: 0.8),
      PlaybackSpeed.fast => (chinese: 0.9, english: 1.0),
    };
  }
}
```

// character_celebration.dart
class CharacterCelebration extends StatefulWidget {
  @override
  _CharacterCelebrationState createState() => _CharacterCelebrationState();
}

class _CharacterCelebrationState extends State<CharacterCelebration>
    with SingleTickerProviderStateMixin {
  late AnimationController _controller;
  late Animation<double> _scaleAnimation;
  late Animation<double> _opacityAnimation;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      duration: Duration(milliseconds: 800),
      vsync: this,
    );

    _scaleAnimation = Tween<double>(begin: 0.5, end: 1.5).animate(
      CurvedAnimation(parent: _controller, curve: Curves.elasticOut),
    );

    _opacityAnimation = Tween<double>(begin: 1.0, end: 0.0).animate(
      CurvedAnimation(parent: _controller, curve: Curves.easeOut),
    );

    _controller.forward();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: _controller,
      builder: (context, child) {
        return Opacity(
          opacity: _opacityAnimation.value,
          child: Transform.scale(
            scale: _scaleAnimation.value,
            child: Icon(
              Icons.star,
              color: Colors.yellow,
              size: 60,
            ),
          ),
        );
      },
    );
  }
}
```
