class AudioCacheService extends GetxService {
  final Map<String, Uint8List> _cache = {};
  final int _maxCacheSize = 50 * 1024 * 1024;  // 50MB

  Future<Uint8List?> getCachedAudio(String key) async {
    return _cache[key];
  }

  Future<void> cacheAudio(String key, Uint8List data) async {
    if (_getCurrentCacheSize() + data.length > _maxCacheSize) {
      _evictOldest();
    }
    _cache[key] = data;
  }

  int _getCurrentCacheSize() {
    return _cache.values.fold(0, (sum, data) => sum + data.length);
  }

class TextToSpeechService {
  final OfflineAudioService _offlineAudio = OfflineAudioService();

  Future<void> speakRegion(InteractiveRegion region) async {
    // 优先使用离线音频
    if (await _offlineAudio.hasOfflineAudio(region.audioUrl)) {
      await _offlineAudio.playOfflineAudio(region.audioUrl);

      // 降级到 TTS
      await _speakWithTTS(region);
    }
  }
}
